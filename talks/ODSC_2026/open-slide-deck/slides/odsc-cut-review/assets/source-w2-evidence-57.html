# Current pagination decision

Status: accepted.

The existing `paginate(items, page, per_page)` API remains public and one-based.
Do not change its signature, return type, validation, or page-number behavior.

Add cursor pagination to `toyapp.pagination` with this public contract:

```python
result = paginate_cursor(items, cursor=None, per_page=2)
result.items
result.next_cursor
result.has_more
```

`paginate_cursor` returns a frozen `CursorPage` dataclass. Cursors are opaque
URL-safe base64 tokens without `=` padding. Before encoding, the token payload
is `v2:<zero-based offset>`. The first call uses an offset of zero. A returned
`next_cursor` points to the first item of the next page and is `None` on the
last page. `has_more` is true exactly when `next_cursor` is not `None`.

Reject malformed cursors, cursors with another version prefix, negative
offsets, and `per_page < 1` with `ValueError`. A valid cursor beyond the end of
the collection returns an empty final page.

Workshop 2 — real evidence tool implementation

Read-only public GitHub access. Inputs owner/repo/pull_number; first 100 files and commits. Field caps: summary 5500 characters, validation 3500, up to five patches 2500 each. No topic-specific filenames in the selection rule. The report describes a 30K bound; this function shows field caps, not a single global cap.

def get_pull_request_evidence_bundle(
    owner: str, repo: str, pull_number: int
) -> dict:
    """Get a compact, citation-ready evidence bundle for explaining one pull request."""
    pull = github_get(f"/repos/{owner}/{repo}/pulls/{pull_number}")
    files = github_get(
        f"/repos/{owner}/{repo}/pulls/{pull_number}/files?per_page=100"
    )
    commits = github_get(
        f"/repos/{owner}/{repo}/pulls/{pull_number}/commits?per_page=100"
    )
    assert isinstance(pull, dict)
    assert isinstance(files, list)
    assert isinstance(commits, list)

    body = pull.get("body") or ""
    validation_start = body.lower().find("automated:")
    validation_end = body.lower().find("## video/screenshots")
    validation = (
        body[validation_start : validation_end if validation_end > validation_start else None]
        if validation_start >= 0
        else ""
    )
    def is_test(item: dict) -> bool:
        filename = (item.get("filename") or "").lower()
        return (
            filename.startswith(("test/", "tests/", "__tests__/"))
            or "/__tests__/" in filename
            or ".test." in filename
            or ".spec." in filename
        )

    production = sorted(
        (item for item in files if not is_test(item)),
        key=lambda item: item.get("changes") or 0,
        reverse=True,
    )
    tests = sorted(
        (item for item in files if is_test(item)),
        key=lambda item: item.get("changes") or 0,
        reverse=True,
    )
    # Keep the selection task-independent: the three largest production changes
    # and two largest test changes provide a compact implementation/validation
    # cross-section without knowing this PR's filenames or subject.
    important = production[:3] + tests[:2]
    return {
        "pull_request": {
            "number": pull.get("number"),
            "title": pull.get("title"),
            "summary": body[:5500],
            "state": pull.get("state"),
            "merged_at": pull.get("merged_at"),
            "html_url": pull.get("html_url"),
            "changed_files": pull.get("changed_files"),
            "commits": pull.get("commits"),
            "additions": pull.get("additions"),
            "deletions": pull.get("deletions"),
        },
        "validation_evidence": validation[:3500],
        "files": [
            {
                "filename": item.get("filename"),
                "status": item.get("status"),
                "changes": item.get("changes"),
                "blob_url": item.get("blob_url"),
            }
            for item in files
        ],
        "important_file_patches": [
            {
                "filename": item.get("filename"),
                "patch": (item.get("patch") or "")[:2500],
                "blob_url": item.get("blob_url"),
            }
            for item in important[:5]
        ],
        "commits": [
            {
                "sha": item.get("sha"),
                "message": ((item.get("commit") or {}).get("message") or "").splitlines()[0],
                "html_url": item.get("html_url"),
            }
            for item in commits
        ],
    }
